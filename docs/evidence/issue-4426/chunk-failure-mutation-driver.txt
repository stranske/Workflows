"""Record each new chunk-failure case against a named, restored mutation."""

import ast
import hashlib
import json
import subprocess
import sys
import time
from pathlib import Path

root = Path.cwd()
evidence = root / "docs/evidence/issue-4426"
source_path = root / "scripts/langchain/pr_verifier.py"
test_path = root / "tests/scripts/test_pr_verifier_ranked_remainder.py"
original = source_path.read_bytes()
source = original.decode()
test_source = test_path.read_text()


def sha(data):
    return hashlib.sha256(data).hexdigest()


def fragment(text, name):
    node = next(n for n in ast.parse(text).body
                if isinstance(n, ast.FunctionDef) and n.name == name)
    return ast.get_source_segment(text, node)


def run(node):
    command = [sys.executable, "-m", "pytest", "-q", node,
               "-m", "not slow", "--tb=short", "-p", "no:cacheprovider"]
    started = time.monotonic()
    result = subprocess.run(command, cwd=root, text=True, capture_output=True, timeout=110)
    return {"command": command, "exit_code": result.returncode,
            "elapsed_seconds": round(time.monotonic() - started, 3),
            "output": result.stdout + result.stderr}


cases = []
for finding in ("PASS", "CONCERNS", "FAIL"):
    cases.append((
        "lost-synthesis-error-" + finding,
        "test_failed_synthesis_keeps_review_credit_and_local_findings[" + finding + "]",
        'error=str(exc),\n                summary=f"Cross-file acceptance reconciliation unavailable: {exc}",',
        'error=None,\n                summary=f"Cross-file acceptance reconciliation unavailable: {exc}",',
    ))
for absent in (False, True):
    cases.append((
        "lost-coverage-disposition-" + str(absent),
        "test_chunk_coverage_rejection_prevents_calls_and_review_credit[" + str(absent) + "]",
        '"CONCERNS" if coverage.required_evidence_absent else "UNVERIFIABLE_INPUT"\n                    ),',
        ('"UNVERIFIABLE_INPUT"' if absent else '"CONCERNS"') + '\n                    ),',
    ))

receipts = []
symbol = fragment(source, "_evaluate_chunked")
try:
    for name, test, old, new in cases:
        assert source.count(old) == 1, (name, source.count(old))
        assert old in symbol
        mutated = source.replace(old, new, 1)
        node = str(test_path.relative_to(root)) + "::" + test
        try:
            source_path.write_text(mutated)
            red = run(node)
        finally:
            source_path.write_bytes(original)
        green = run(node)
        failures = [line for line in red["output"].splitlines() if line.startswith("FAILED ")]
        assert red["exit_code"] == 1 and "1 failed" in red["output"], red
        assert any(node in line for line in failures), red
        assert green["exit_code"] == 0 and "1 passed" in green["output"], green
        receipts.append({"id": name, "source": str(source_path.relative_to(root)),
                         "symbol": "_evaluate_chunked", "nodeid": node,
                         "original_snippet": old, "mutant_snippet": new,
                         "restored_sha256": sha(original),
                         "mutant_sha256": sha(mutated.encode()),
                         "mutant_symbol_sha256": sha(symbol.replace(old, new, 1).encode()),
                         "test_sha256": sha(test_path.read_bytes()),
                         "red_named_failure": failures, "red": red, "green": green})
        print(name, "red=1 restored=0", flush=True)
finally:
    source_path.write_bytes(original)

record = {
    "checked_on": "2026-11-07", "source_restored": source_path.read_bytes() == original,
    "source_sha256": sha(original), "recorded_symbol": symbol,
    "symbol_sha256": sha(symbol.encode()),
    "test_sha256": sha(test_path.read_bytes()), "recorded_test_source": test_source,
    "receipts": receipts,
}
assert record["source_restored"]
(evidence / "chunk-failure-mutations.json").write_text(json.dumps(record, indent=2) + "\n")
