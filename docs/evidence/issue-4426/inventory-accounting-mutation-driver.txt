"""Retain deliberate-break receipts for complete Gate inventory accounting."""

import hashlib
import json
import subprocess
import sys
from pathlib import Path

root = Path(__file__).resolve().parents[3]
source = root / "scripts/compare_gate_coverage.py"
test = root / "tests/scripts/test_reconcile_issue_4426_coverage.py"
original = source.read_bytes()
cases = [
    (
        "summary_completeness_and_integer_types",
        "if type(summary[field]) is not int or summary[field] != expected:",
        "if False:",
        "test_source_mapping_rejects_incomplete_or_mistyped_summary",
    ),
    (
        "removed_code_is_not_a_mapped_gain",
        'counts["newly_covered"] - counts["newly_missing"]',
        'counts["newly_covered"] - counts["newly_missing"] + counts["unmatched_baseline_missing"]',
        "test_source_mapping_accounts_for_missing_units_without_crediting_removed_code",
    ),
    (
        "retained_gate_branch_inventory_is_preserved",
        "return missing_lines | lines, missing_arcs | arcs, missing_lines, missing_arcs",
        "return missing_lines | lines, set(), missing_lines, missing_arcs",
        "test_source_mapping_accepts_complete_retained_gate_inventories",
    ),
]


def run(selection):
    command = [
        sys.executable, "-m", "pytest", "-q", str(test.relative_to(root)),
        "-m", "not slow", "-k", selection, "--tb=short",
    ]
    result = subprocess.run(command, cwd=root, capture_output=True, text=True, timeout=110)
    return {"command": command, "exit_code": result.returncode, "output": result.stdout + result.stderr}


receipts = []
try:
    for name, needle, replacement, selection in cases:
        assert original.decode().count(needle) == 1, name
        source.write_text(original.decode().replace(needle, replacement))
        red = run(selection)
        source.write_bytes(original)
        green = run(selection)
        receipts.append({
            "name": name, "needle": needle, "replacement": replacement,
            "red": red, "restored_green": green,
        })
        assert red["exit_code"] == 1 and "FAILED " in red["output"], name
        assert green["exit_code"] == 0 and "passed" in green["output"], name
finally:
    source.write_bytes(original)
    output = {
        "schema": "workflows.issue-4663-inventory-accounting-mutations/v1",
        "source_sha256": hashlib.sha256(source.read_bytes()).hexdigest(),
        "test_sha256": hashlib.sha256(test.read_bytes()).hexdigest(),
        "cases": receipts,
    }
    (Path(__file__).parent / "inventory-accounting-mutations.json").write_text(
        json.dumps(output, indent=2) + "\n"
    )
print(f"{len(receipts)} named deliberate breaks failed and restored selections passed")
