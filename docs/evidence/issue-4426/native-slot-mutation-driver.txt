"""Record native-slot regressions against immutable symbol/test evidence."""

import ast
import hashlib
import json
import subprocess
import sys
from pathlib import Path

root = Path.cwd()
evidence = root / "docs/evidence/issue-4426"
source_path = root / "scripts/langchain/pr_verifier.py"
test_path = root / "tests/scripts/test_pr_verifier_ranked_remainder.py"
original = source_path.read_bytes()
source = original.decode()

def fragment(text, name):
    node = next(n for n in ast.parse(text).body if isinstance(n, ast.FunctionDef) and n.name == name)
    return ast.get_source_segment(text, node)

def run(test):
    command = [sys.executable, "-m", "pytest", "-q", str(test_path.relative_to(root)) + "::" + test,
               "-m", "not slow", "--tb=short", "-p", "no:cacheprovider"]
    result = subprocess.run(command, cwd=root, text=True, capture_output=True, timeout=110)
    return {"command": command, "exit_code": result.returncode, "output": result.stdout + result.stderr}

cases = [
    ("accept-mismatched-native-slot", "_semantic_slots_and_clients",
     "test_semantic_slot_mismatch_rejects_native_contract",
     'if contract["provider"] != provider or contract["model"] != model:', 'if False:'),
    ("accept-sdk-retries", "_semantic_slots_and_clients",
     "test_semantic_slots_require_zero_native_sdk_retries",
     'if getattr(native_sdk, "max_retries", None) != 0:', 'if False:'),
    ("skip-paired-sdk-check", "_semantic_slots_and_clients",
     "test_semantic_slots_preserve_overrides_trial_sdk_checks_and_provenance",
     '            trial.check_sdk(native_sdk)', '            pass'),
    ("authenticate-synthetic-slot", "_semantic_slots_and_clients",
     "test_semantic_slots_preserve_overrides_trial_sdk_checks_and_provenance",
     'authentic = authentic and contract.get("provenance") != "synthetic-unit-test-only"',
     'authentic = True'),
    ("drop-package-chain-depth", "_semantic_package_prompt",
     "test_semantic_package_prompt_retains_bytes_selected_obligations_and_chain_depth",
     'prompt += "\\n" + CHAIN_DEPTH_ADDENDUM.format(depth=_get_chain_depth())', 'prompt += ""'),
    ("include-unassigned-obligations", "_semantic_package_prompt",
     "test_semantic_package_prompt_retains_bytes_selected_obligations_and_chain_depth",
     'if clause["id"] in package["clause_ids"]', 'if True'),
]
symbols = {name: fragment(source, name) for _, name, *_ in cases}
receipts = []
try:
    for name, symbol_name, test, old, new in cases:
        symbol = symbols[symbol_name]
        assert symbol.count(old) == 1, (name, symbol.count(old))
        mutated_symbol = symbol.replace(old, new, 1)
        mutated = source.replace(symbol, mutated_symbol, 1)
        try:
            source_path.write_text(mutated)
            red = run(test)
        finally:
            source_path.write_bytes(original)
        green = run(test)
        failures = [line for line in red["output"].splitlines() if line.startswith("FAILED ")]
        assert red["exit_code"] == 1 and failures, red
        assert all(test in line for line in failures), red
        assert green["exit_code"] == 0, green
        receipts.append({"id": name, "symbol": symbol_name, "test": test,
                         "original_snippet": old, "mutant_snippet": new,
                         "mutant_sha256": hashlib.sha256(mutated.encode()).hexdigest(),
                         "named_failures": failures, "red": red, "green": green})
        print(name, "red=1 restored=0", flush=True)
finally:
    source_path.write_bytes(original)

test_source = test_path.read_text()
record = {
    "checked_on": "2026-11-07", "source_path": str(source_path.relative_to(root)),
    "source_restored": source_path.read_bytes() == original,
    "source_sha256": hashlib.sha256(original).hexdigest(),
    "recorded_symbols": symbols,
    "symbol_sha256": {name: hashlib.sha256(text.encode()).hexdigest() for name, text in symbols.items()},
    "test_path": str(test_path.relative_to(root)),
    "test_sha256": hashlib.sha256(test_path.read_bytes()).hexdigest(),
    "recorded_tests": {test: fragment(test_source, test) for _, _, test, *_ in cases},
    "recorded_fixture": fragment(test_source, "native_slots"),
    "receipts": receipts,
}
assert record["source_restored"]
(evidence / "native-slot-mutations.json").write_text(json.dumps(record, indent=2) + "\n")
