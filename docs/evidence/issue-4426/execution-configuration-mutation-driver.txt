"""Record discriminating configuration tests and restore source after each mutant."""

import ast
import hashlib
import json
import subprocess
import sys
from pathlib import Path

root = Path.cwd()
evidence = root / "docs/evidence/issue-4426"
source_path = root / "scripts/reconcile_issue_4426_coverage.py"
test_path = root / "tests/scripts/test_reconcile_issue_4426_coverage.py"
original = source_path.read_bytes()
source = original.decode()
test_source = test_path.read_text()


def fragment(text, name):
    node = next(n for n in ast.parse(text).body
                if isinstance(n, ast.FunctionDef) and n.name == name)
    return ast.get_source_segment(text, node)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def run(test):
    command = [sys.executable, "-m", "pytest", "-q",
               str(test_path.relative_to(root)) + "::" + test,
               "-m", "not slow", "--tb=short", "-p", "no:cacheprovider"]
    result = subprocess.run(command, cwd=root, capture_output=True, text=True, timeout=110)
    return {"command": command, "exit_code": result.returncode,
            "output": result.stdout + result.stderr}


cases = [
    ("ignore-report-exclusions", "_execution_settings",
     "test_execution_reconciliation_records_policy_drift_without_rewriting_history",
     '"coverage": tools["coverage"],',
     '"coverage": {"run": tools["coverage"]["run"]},'),
    ("lose-removed-configuration", "_configuration_differences",
     "test_execution_difference_keeps_removed_exclusions_and_changed_source_selection",
     'sorted(before.keys() | after.keys())', 'sorted(after.keys())'),
    ("credit-matching-settings-without-execution", "reconcile_execution_configurations",
     "test_matching_execution_settings_do_not_establish_attribution",
     '"attributable_round_reduction": None,', '"attributable_round_reduction": 0,'),
    ("accept-substituted-execution-evidence", "reconcile_execution_configurations",
     "test_execution_reconciliation_rejects_substituted_record",
     'if hashlib.sha256(data).hexdigest() != EXECUTION_INPUTS_SHA256:', 'if False:'),
    ("accept-ambiguous-observed-runtime", "_execution_settings",
     "test_execution_settings_require_unambiguous_observed_runtime",
     'if len(matches) != 1:', 'if False:'),
]
symbols = {symbol: fragment(source, symbol) for _, symbol, *_ in cases}
receipts = []
try:
    for name, symbol_name, test, old, new in cases:
        symbol = symbols[symbol_name]
        assert symbol.count(old) == 1, (name, symbol.count(old))
        mutant_symbol = symbol.replace(old, new, 1)
        mutant = source.replace(symbol, mutant_symbol, 1)
        try:
            source_path.write_text(mutant)
            red = run(test)
        finally:
            source_path.write_bytes(original)
        green = run(test)
        failures = [line for line in red["output"].splitlines() if line.startswith("FAILED ")]
        assert red["exit_code"] == 1 and failures, red
        assert all(test in line for line in failures), failures
        assert len(failures) == (2 if test.endswith("observed_runtime") else 1), red
        assert green["exit_code"] == 0, green
        receipts.append({"id": name, "symbol": symbol_name, "test": test,
                         "original_snippet": old, "mutant_snippet": new,
                         "mutant_sha256": sha(mutant.encode()),
                         "red_named_failures": failures, "red": red, "green": green})
        print(name, "mutant=1 restored=0", flush=True)
finally:
    source_path.write_bytes(original)

record = {
    "checked_on": "2026-11-07",
    "scope": "Execution configuration comparison only; no round-wide coverage credit",
    "source_path": str(source_path.relative_to(root)),
    "source_sha256": sha(original),
    "recorded_symbols": symbols,
    "symbol_sha256": {name: sha(text.encode()) for name, text in symbols.items()},
    "test_path": str(test_path.relative_to(root)),
    "test_sha256": sha(test_path.read_bytes()),
    "recorded_tests": {test: fragment(test_source, test) for _, _, test, *_ in cases},
    "source_restored": source_path.read_bytes() == original,
    "receipts": receipts,
}
assert record["source_restored"]
(evidence / "execution-configuration-mutations.json").write_text(json.dumps(record, indent=2) + "\n")
