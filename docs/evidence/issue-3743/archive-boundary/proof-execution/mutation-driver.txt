import difflib
import hashlib
import json
import os
import subprocess
import sys
import xml.etree.ElementTree as ET
from pathlib import Path

root = Path.cwd()
evidence = root / 'docs/evidence/issue-3743/archive-boundary/proof-execution'
paths = [root / 'scripts/check_deliberate_break.py', root / 'templates/consumer-repo/scripts/check_deliberate_break.py']
original = {path: path.read_bytes() for path in paths}
environment = os.environ.copy()
for name in ['GITHUB_OUTPUT', 'GITHUB_STEP_SUMMARY']:
    environment.pop(name, None)
manifest = {'interpreter': sys.executable, 'runs': {}, 'mutations': {}}

def run(name, selection, expected_failed, expected_passed):
    command = [sys.executable, '-m', 'pytest', 'tests/scripts/test_check_deliberate_break_archive.py', '-vv', '--tb=short', '-o', 'addopts=', '-m', 'not slow', '-k', selection, '--junitxml=' + str(evidence / (name + '.xml'))]
    result = subprocess.run(command, cwd=root, env=environment, text=True, capture_output=True)
    output = 'Command: ' + ' '.join(command) + '\nExit: ' + str(result.returncode) + '\n' + result.stdout + result.stderr
    (evidence / (name + '.txt')).write_text('\n'.join(line.rstrip() for line in output.splitlines()) + '\n')
    cases = ET.parse(evidence / (name + '.xml')).findall('.//testcase')
    failed = [case.attrib['name'] for case in cases if case.find('failure') is not None]
    passed = [case.attrib['name'] for case in cases if len(case) == 0]
    manifest['runs'][name] = {'command': command, 'exit': result.returncode, 'failed': failed, 'passed': passed}
    print(name, result.returncode, len(failed), len(passed), flush=True)
    assert len(failed) == expected_failed and len(passed) == expected_passed, output
    assert result.returncode == (1 if expected_failed else 0), output

def restore():
    for path, source in original.items():
        path.write_bytes(source)
        assert path.read_bytes() == source

def mutate(name, before, after):
    changes = []
    for path, source in original.items():
        text = source.decode()
        assert text.count(before) == 1
        changed = text.replace(before, after)
        path.write_text(changed)
        changes.extend(difflib.unified_diff(text.splitlines(), changed.splitlines(), fromfile=str(path.relative_to(root)), tofile=str(path.relative_to(root)), n=2, lineterm=''))
    (evidence / (name + '.diff')).write_text('\n'.join(changes) + '\n')
    manifest['mutations'][name] = {'before': before, 'after': after}

experiments = [
    ('execute-mode', 'resolved.chmod(member.mode & 0o777)', 'resolved.chmod(0o600)', 'preserves_executable', 2),
    ('external-symlink', 'if not member.isfile():', 'if not (member.isfile() or member.issym()):', 'external_symlink', 2),
    ('archive-escape', 'raise ValueError(f"unsafe archive path: {member.name}")', 'pass  # deliberate removal of containment guard', 'archive_escape', 4),
    ('unexecuted-head', 'head_run = _run_with_runtime_deps(spec.command, repo)', 'head_run = subprocess.CompletedProcess(spec.command, 0, "", "")', '', 8),
    ('unexecuted-base', 'base_run = _run_with_runtime_deps(spec.command, base_dir)', 'base_run = subprocess.CompletedProcess(spec.command, 1, "archive-proof-base\\nregular-base", "")', 'preserves_executable or external_symlink', 4),
]
try:
    for name, before, after, selected, count in experiments:
        try:
            mutate(name, before, after)
            run(name + '-RED', selected, count, 0)
        finally:
            restore()
        manifest['mutations'][name]['restored_sha256'] = {str(path.relative_to(root)): hashlib.sha256(path.read_bytes()).hexdigest() for path in paths}
        run(name + '-GREEN', selected, 0, count)
finally:
    restore()
    manifest['sources'] = {str(path.relative_to(root)): {'before_sha256': hashlib.sha256(source).hexdigest(), 'after_sha256': hashlib.sha256(path.read_bytes()).hexdigest(), 'byte_identical': path.read_bytes() == source} for path, source in original.items()}
    manifest['test_sha256'] = hashlib.sha256((root / 'tests/scripts/test_check_deliberate_break_archive.py').read_bytes()).hexdigest()
    (evidence / 'mutation-manifest.json').write_text(json.dumps(manifest, indent=2) + '\n')
