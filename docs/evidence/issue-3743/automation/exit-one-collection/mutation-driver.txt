import difflib
import hashlib
import json
import os
import subprocess
import sys
import xml.etree.ElementTree as ET
from pathlib import Path

root = Path.cwd()
evidence = root / "docs/evidence/issue-3743/automation/exit-one-collection"
paths = [root / "scripts/check_deliberate_break.py", root / "templates/consumer-repo/scripts/check_deliberate_break.py"]
original = {path: path.read_bytes() for path in paths}
environment = os.environ.copy()
for name in ["GITHUB_OUTPUT", "GITHUB_STEP_SUMMARY"]:
    environment.pop(name, None)
selection = "test_base_pytest_must_execute_a_test_before_counting_as_red and -1-"
controls = "test_base_missing_import_inside_a_test_still_proves_red"
manifest = {"interpreter": sys.executable, "runs": {}}

def run(name, selected, expected_failed, expected_passed):
    command = [sys.executable, "-m", "pytest", "tests/scripts/test_check_deliberate_break.py", "-vv", "--tb=short", "-o", "addopts=", "-m", "not slow", "-k", selected, "--junitxml=" + str(evidence / (name + ".xml"))]
    result = subprocess.run(command, cwd=root, env=environment, text=True, capture_output=True)
    output = "Command: " + " ".join(command) + "\nExit: " + str(result.returncode) + "\n" + result.stdout + result.stderr
    (evidence / (name + ".txt")).write_text("\n".join(line.rstrip() for line in output.splitlines()) + "\n")
    cases = ET.parse(evidence / (name + ".xml")).findall(".//testcase")
    receipt = evidence / (name + ".xml")
    receipt.write_text("\n".join(line.rstrip() for line in receipt.read_text().splitlines()) + "\n")
    failed = [case.attrib["name"] for case in cases if case.find("failure") is not None]
    passed = [case.attrib["name"] for case in cases if len(case) == 0]
    manifest["runs"][name] = {"command": command, "exit": result.returncode, "failed": failed, "passed": passed}
    print(name, result.returncode, len(failed), len(passed), flush=True)
    assert len(failed) == expected_failed and len(passed) == expected_passed, output
    assert result.returncode == (1 if expected_failed else 0), output

def restore():
    for path, source in original.items():
        path.write_bytes(source)
        assert path.read_bytes() == source

try:
    changes = []
    for path, source in original.items():
        relative = str(path.relative_to(root))
        prior = subprocess.check_output(["git", "show", "ed99df79:" + relative], cwd=root)
        assert prior != source
        path.write_bytes(prior)
        changes.extend(difflib.unified_diff(source.decode().splitlines(), prior.decode().splitlines(), fromfile=relative, tofile=relative, n=0, lineterm=""))
    (evidence / "prior-implementation.diff").write_text("\n".join(changes) + "\n")
    run("prior-implementation-RED", selection, 4, 0)
finally:
    restore()
    manifest["restored_sources"] = {str(path.relative_to(root)): {"before": hashlib.sha256(source).hexdigest(), "after": hashlib.sha256(path.read_bytes()).hexdigest(), "byte_identical": path.read_bytes() == source} for path, source in original.items()}
    manifest["test_sha256"] = hashlib.sha256((root / "tests/scripts/test_check_deliberate_break.py").read_bytes()).hexdigest()
    (evidence / "mutation-manifest.json").write_text(json.dumps(manifest, indent=2) + "\n")
run("restored-GREEN", "(" + selection + ") or " + controls, 0, 12)
(evidence / "mutation-manifest.json").write_text(json.dumps(manifest, indent=2) + "\n")
